cmake_minimum_required(VERSION 3.24)
project(ritmo_native C)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
option(RN_SANITIZE "Address and undefined behavior sanitizers" OFF)
option(RN_HEADLESS_DRIVERS "Build SDL with only offscreen/dummy video drivers" OFF)
if(RN_SANITIZE AND NOT MSVC)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()
add_subdirectory(vendor/quickjs EXCLUDE_FROM_ALL)
include(FetchContent)
if(ANDROID)
  set(SDL_SHARED ON CACHE BOOL "" FORCE)
  set(SDL_STATIC OFF CACHE BOOL "" FORCE)
else()
  set(SDL_SHARED OFF CACHE BOOL "" FORCE)
  set(SDL_STATIC ON CACHE BOOL "" FORCE)
endif()
set(SDL_TEST_LIBRARY OFF CACHE BOOL "" FORCE)
set(SDL_TESTS OFF CACHE BOOL "" FORCE)
if(RN_HEADLESS_DRIVERS)
  set(SDL_UNIX_CONSOLE_BUILD ON CACHE BOOL "" FORCE)
  set(SDL_X11 OFF CACHE BOOL "" FORCE)
  set(SDL_WAYLAND OFF CACHE BOOL "" FORCE)
  set(SDL_KMSDRM OFF CACHE BOOL "" FORCE)
endif()
FetchContent_Declare(SDL3 URL https://www.libsdl.org/release/SDL3-3.2.28.tar.gz URL_HASH SHA256=1330671214d146f8aeb1ed399fc3e081873cdb38b5189d1f8bb6ab15bbc04211 DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
if(DEFINED ENV{FETCHCONTENT_SOURCE_DIR_SDL3})
  set(FETCHCONTENT_SOURCE_DIR_SDL3 "$ENV{FETCHCONTENT_SOURCE_DIR_SDL3}")
endif()
FetchContent_MakeAvailable(SDL3)
if(NOT RN_BUNDLE)
  message(FATAL_ERROR "Set RN_BUNDLE to the generated bundle.c; use ritmo-native build")
endif()
add_library(rn_stb STATIC vendor/stb_image.c)
set(RN_RUNTIME_SOURCES runtime/rn_main.c runtime/rn_platform_entry_sdl3.c runtime/rn_engine_qjs.c runtime/rn_gfx.c runtime/rn_kernel.c runtime/rn_platform_sdl3.c "${RN_BUNDLE}" "${RN_KERNELS}")
if(ANDROID)
  add_library(ritmo_app SHARED ${RN_RUNTIME_SOURCES})
  set_target_properties(ritmo_app PROPERTIES OUTPUT_NAME main)
else()
  add_executable(ritmo_app ${RN_RUNTIME_SOURCES})
endif()
target_compile_definitions(ritmo_app PRIVATE RN_ENTRY=rn_app_main)
target_include_directories(ritmo_app PRIVATE runtime vendor)
target_link_libraries(ritmo_app PRIVATE qjs rn_stb SDL3::SDL3)
if(RN_APP_NAME)
  target_compile_definitions(ritmo_app PRIVATE RN_APP_NAME="${RN_APP_NAME}")
endif()
if(RN_WIDTH)
  target_compile_definitions(ritmo_app PRIVATE RN_WIDTH=${RN_WIDTH} RN_HEIGHT=${RN_HEIGHT})
endif()
if(MSVC)
  target_compile_options(ritmo_app PRIVATE /W4 /WX /fp:precise)
else()
  target_compile_options(ritmo_app PRIVATE -Wall -Wextra -Werror -ffp-contract=off)
endif()
if(APPLE)
  set_target_properties(ritmo_app PROPERTIES MACOSX_BUNDLE TRUE MACOSX_BUNDLE_GUI_IDENTIFIER "${RN_APP_ID}" MACOSX_BUNDLE_BUNDLE_VERSION "1" MACOSX_BUNDLE_SHORT_VERSION_STRING "${RN_APP_VERSION}")
  if(RN_IOS_PLIST)
    set_target_properties(ritmo_app PROPERTIES MACOSX_BUNDLE_INFO_PLIST "${RN_IOS_PLIST}")
  endif()
  if(RN_ASSETS)
    file(GLOB_RECURSE RN_ASSET_FILES CONFIGURE_DEPENDS "${RN_ASSETS}/*")
    foreach(RN_ASSET ${RN_ASSET_FILES})
      file(RELATIVE_PATH RN_ASSET_RELATIVE "${RN_ASSETS}" "${RN_ASSET}")
      get_filename_component(RN_ASSET_DIR "${RN_ASSET_RELATIVE}" DIRECTORY)
      set_source_files_properties("${RN_ASSET}" PROPERTIES MACOSX_PACKAGE_LOCATION "Resources/assets/${RN_ASSET_DIR}")
    endforeach()
    target_sources(ritmo_app PRIVATE ${RN_ASSET_FILES})
  endif()
endif()
include(CTest)
add_test(NAME native_help COMMAND ritmo_app --help)

if(BUILD_TESTING AND NOT ANDROID AND NOT IOS AND EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/test/rn_kernel.test.c")
  add_executable(rn_kernel_test test/rn_kernel.test.c runtime/rn_kernel.c)
  target_include_directories(rn_kernel_test PRIVATE runtime)
  if(NOT MSVC)
    target_link_libraries(rn_kernel_test PRIVATE m)
    target_compile_options(rn_kernel_test PRIVATE -Wall -Wextra -Werror -ffp-contract=off)
  endif()
  add_test(NAME kernel_semantics COMMAND rn_kernel_test)
  add_executable(rn_gfx_test test/rn_gfx.test.c runtime/rn_gfx.c runtime/rn_platform_sdl3.c)
  target_include_directories(rn_gfx_test PRIVATE runtime vendor)
  target_link_libraries(rn_gfx_test PRIVATE rn_stb SDL3::SDL3)
  if(NOT MSVC)
    target_compile_options(rn_gfx_test PRIVATE -Wall -Wextra -Werror)
  endif()
  add_test(NAME gfx_screenshot COMMAND rn_gfx_test "${CMAKE_CURRENT_SOURCE_DIR}/test/fixtures/scene.bmp" "${CMAKE_CURRENT_BINARY_DIR}/scene.bmp" "${CMAKE_CURRENT_SOURCE_DIR}/test/fixtures/pixels.png")
endif()
