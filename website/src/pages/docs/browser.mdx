## The loop

Fixed updates whatever the frame rate, then one `render`. Long frames are
clamped.

```ts twoslash title="main.ts"
import { Clock, Descriptor, Schema } from "@typeonce/ritmo"
const Game = Schema.bind(Schema.fragment({ components: { Tag: Descriptor.Component<{}>()("Example/Tag") } }))
const update = Game.Schedule()
const runtime = Game.Runtime.make({ services: Game.Runtime.services() }).value!
declare const draw: (alpha: number) => void
// ---cut---
import * as FixedLoop from "@typeonce/ritmo/FixedLoop"
import { BrowserTickSource } from "@typeonce/ritmo-browser"

// Turns animation frames into fixed steps
const loop = FixedLoop.start({
  source: BrowserTickSource.make(window),
  step: Clock.hz(60), // [!code highlight]

  // A tab back from the background does not run a minute of game at once
  maxFrameSeconds: 0.25,
  maxStepsPerFrame: 5,

  update: () => runtime.tick(update), // [!code highlight]
  render: (frame) => draw(frame.alpha), // [!code highlight]
  onFailure: (failure) => console.error(failure)
})
```

A renderer with its own ticker plugs in through `FixedLoop.source`.

## Input

Systems see **actions**, never keys. A button action is pressed and held
(jump); a vector action is a direction of length at most 1 (move). Each
device binds them, and the devices merge into one source.

```ts twoslash title="input.ts"
import { Descriptor, Schema } from "@typeonce/ritmo"
// ---cut---
import {
  Controls,
  Gamepad,
  InputCapture,
  Keyboard
} from "@typeonce/ritmo-browser"

// What the player can do, whatever the device
const Actions = Controls.actions({ buttons: ["jump"], vectors: ["move"] }) // [!code highlight]

// How each device triggers it
const keys = {
  move: Keyboard.vector({
    up: ["ArrowUp", "w"],
    down: ["ArrowDown", "s"],
    left: ["ArrowLeft", "a"],
    right: ["ArrowRight", "d"]
  }),
  jump: [" "]
} as const
const buttons = {
  move: Gamepad.vector({
    stick: "left",
    up: ["dpadUp"],
    down: ["dpadDown"],
    left: ["dpadLeft"],
    right: ["dpadRight"]
  }),
  jump: ["south"]
} as const

// Keyboard and gamepad merged into one source
const makeControls = (devices: {
  readonly keyboard: Keyboard.Actions<typeof keys>
  readonly gamepad: Gamepad.Actions<typeof buttons>
}) => Actions.merge(devices)

// The input source is a service: the merge's snapshot() only, so a replay
// or the fuzzer can provide it too. Each update copies the snapshot into a
// resource that systems read.
type ControlsSource = InputCapture.SourceOf<ReturnType<typeof makeControls>>
const ControlsInput = Descriptor.Service<ControlsSource>()("Example/Controls")
type Snapshot = InputCapture.SnapshotOf<typeof ControlsInput>
const Input = Descriptor.TransientResource<Snapshot>()("Example/Input")
```

One snapshot per update, so every system sees the same input:

```ts twoslash title="jump.ts"
import { Descriptor, Schema } from "@typeonce/ritmo"
import { Controls, Gamepad, InputCapture, Keyboard } from "@typeonce/ritmo-browser"
const Actions = Controls.actions({ buttons: ["jump"], vectors: ["move"] })
const keys = { move: Keyboard.vector({ up: ["w"], down: ["s"], left: ["a"], right: ["d"] }), jump: [" "] } as const
const buttons = { move: Gamepad.vector({ stick: "left" }), jump: ["south"] } as const
const makeControls = (devices: { readonly keyboard: Keyboard.Actions<typeof keys>; readonly gamepad: Gamepad.Actions<typeof buttons> }) => Actions.merge(devices)
const ControlsInput = Descriptor.Service<InputCapture.SourceOf<ReturnType<typeof makeControls>>>()("Example/Controls")
const Input = Descriptor.TransientResource<InputCapture.SnapshotOf<typeof ControlsInput>>()("Example/Input")
const Game = Schema.bind(Schema.fragment({ resources: { Input } }))
// ---cut---
const CaptureInput = InputCapture.system(Game, {
  name: "Example/CaptureInput",
  source: ControlsInput,
  resource: Input
})

const Jump = Game.System(
  "Example/Jump",
  { resources: { input: Game.System.readResource(Input) } },
  ({ resources }) => {
    const input = resources.input.get()
    if (input.actions.jump.pressed) {
      //                  ^?
    }
    // Keys, d-pad, or stick: length at most 1, diagonals not faster
    const move = input.vectors.move
    //                         ^?
  }
)
```

```ts
const keyboard = Keyboard.actions(window, keys) // in the page
const keyboard = Keyboard.scripted(keys, [
  { frame: 0, press: ["jump", "move.right"] }
]) // in a test
```

`Pointer` and `Touch` work the same way.

Only the capture takes snapshots: each `snapshot()` starts a new press
window, so a second caller would steal presses. Calling it through
`Game.System.service` does not compile; a system that drains a source itself
declares it with `Game.System.consumeService`, and `describe()` warns
(`service-consumed-twice`) when one source is drained in two places.

### Local multiplayer

Players are slots. No device belongs to a player until someone presses
join on it: the first device to join is `p1`, the next `p2`. Two
`Keyboard.actions` over different keys are two keyboard players, and
`Gamepad.pads` lets every pad join on its own.

```ts twoslash title="players.ts"
import { Descriptor } from "@typeonce/ritmo"
import { Controls, Gamepad, InputCapture, Keyboard } from "@typeonce/ritmo-browser"
// ---cut---
const Actions = Controls.actions({
  buttons: ["join", "back", "jump"],
  vectors: ["move"]
})
// Any button joins (that press does nothing else);
// back leaves while joining is open
const Players = Controls.players(Actions, { // [!code highlight]
  slots: ["p1", "p2"], // [!code highlight]
  join: "any", // [!code highlight]
  leave: "back" // [!code highlight]
}) // [!code highlight]

const wasd = {
  move: Keyboard.vector({ up: ["w"], down: ["s"], left: ["a"], right: ["d"] }),
  join: [" "],
  back: ["q"],
  jump: [" "]
} as const
const arrows = {
  move: Keyboard.vector({
    up: ["ArrowUp"],
    down: ["ArrowDown"],
    left: ["ArrowLeft"],
    right: ["ArrowRight"]
  }),
  join: ["Enter"],
  back: ["Backspace"],
  jump: ["Enter"]
} as const
const pad = {
  move: Gamepad.vector({ stick: "left" }),
  join: ["south"],
  back: ["east"],
  jump: ["south"]
} as const

const makePlayers = (devices: {
  readonly left: Keyboard.Actions<typeof wasd>
  readonly right: Keyboard.Actions<typeof arrows>
  readonly pads: Gamepad.Pads<typeof pad>
}) => Players.pool(devices)

// The service is the pool's snapshot() only, so replays can provide it too
type PlayersSource = InputCapture.SourceOf<ReturnType<typeof makePlayers>>
const PlayersInput = Descriptor.Service<PlayersSource>()("Example/Players")
type Snapshot = InputCapture.SnapshotOf<typeof PlayersInput>
const Input = Descriptor.TransientResource<Snapshot>()("Example/Input")

// In a system: input = resources.input.get()
declare const input: Snapshot
if (input.p2.event === "joined") {
  // spawn the second player
}
const move = input.p2.vectors.move
if (input.p2.device === "pads") {
  const stick = input.p2.devices.pads.left
  //                             ^?
}
```

A system opens joining with `setJoining(true)`, once for drop-in play or
only on a join screen. It frees a slot with `leave("p2")`, from a pause
menu for example.

A player who is in from the start is seated from a setup system, without a
press: `seat("p1", "right")` for a pool name, or `seat("p1", { pads: 0 })`
for the first pad. A taken slot or device comes back as
`{ ok: false, reason }`.

A pad that disconnects keeps its slot (`connected: false`) and returns to
it. The town on the home page plays this way: player 1 is seated on the
arrows, and WASD or a gamepad joins player 2.

## Rendering

One node per entity: created when it appears, updated when it moves,
destroyed when it goes. You say what a node is.

```ts twoslash title="render.ts"
import { Descriptor, Schema } from "@typeonce/ritmo"
const Position = Descriptor.Component<{ x: number; y: number }>()("TopDown/Position")
const Collider = Descriptor.Component<{ width: number; height: number }>()("TopDown/Collider")
const Wall = Descriptor.Component<{}>()("TopDown/Wall")
const Game = Schema.bind(Schema.fragment({ components: { Position, Collider, Wall } }))
// ---cut---
import { RenderSync } from "@typeonce/ritmo-render"
import type { NodeRegistry } from "@typeonce/ritmo-render"
import { Graphics } from "pixi.js"

const WallNodes =
  Descriptor.Service<NodeRegistry.NodeRegistry<Graphics>>()("TopDown/WallNodes")

const RenderWalls = RenderSync.system(Game, {
  name: "TopDown/RenderWalls",
  renderable: Wall,
  transform: Position,
  registry: WallNodes,
  select: { collider: Game.Query.read(Collider) },
  create: ({ data }) => {
    // [!code highlight]
    const { width, height } = data.collider.get()
    return new Graphics()
      .rect(-width / 2, -height / 2, width, height)
      .fill(0x262a31)
  },
  apply: (node, { transform }) => node.position.set(transform.x, transform.y) // [!code highlight]
})
```

The system needs a `NodeRegistry`, not Pixi. The page picks the renderer:

```ts title="main.ts"
Game.Runtime.service(WallNodes, PixiNodeRegistry.make(wallLayer))
```

Any renderer works: `NodeRegistry.make({ attach, detach })`. Full host:
[`website/src/demos/top-down`](https://github.com/typeonce-dev/ritmo/tree/main/website/src/demos/top-down).

A part that moves with an entity, like a turret on a tank, is a child entity in a
`Descriptor.Hierarchy`. Pass it as `hierarchy: Attached` and the child's node is added under
its parent's node, so its `Position` is relative to the parent and the renderer composes the two.
