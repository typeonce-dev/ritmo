import { Clock, Descriptor, Schema } from "@typeonce/ritmo"
const Time = Clock.resource("Arena/Time", Clock.hz(60))
const Position = Descriptor.Component<{ x: number; y: number }>()("Arena/Position")
const Velocity = Descriptor.Component<{ x: number; y: number }>()("Arena/Velocity")
const Root = Schema.defineRoot("Arena")
const Game = Schema.bind(Schema.fragment({ components: { Position, Velocity }, resources: { Time } }), Root)
// ---cut---
// What this system reads and writes, declared up front
const Moving = Game.Query({
  selection: {
    position: Game.Query.write(Position), // [!code highlight]
    velocity: Game.Query.read(Velocity) // [!code highlight]
  }
})

export const Move = Game.System(
  "Arena/Move",
  {
    queries: { moving: Moving },
    resources: { time: Game.System.readResource(Time) }
  },

  // It receives exactly that, typed, and nothing else
  ({ queries, resources }) => {
    const { dt } = resources.time.get()

    const { data } = queries.moving
    for (const entity of queries.moving.each()) {
      const p = data.position.get(entity)
      const v = data.velocity.get(entity)

      data.position.set(entity, { x: p.x + v.x * dt, y: p.y + v.y * dt })
    }
  }
)
